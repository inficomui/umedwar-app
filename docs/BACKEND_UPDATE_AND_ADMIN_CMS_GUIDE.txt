3.1 Database Schema for `pramukh_sankalp` Table
----------------------------------------------------------------------------------------
This table stores both the compact card summary (for mobile home screen / vision carousel)
and the complete Word-style rich document HTML (for deep-dive resolution detail screen):

```sql
CREATE TABLE IF NOT EXISTS `pramukh_sankalp` (
  `id` INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  `title` VARCHAR(255) NOT NULL COMMENT 'मराठी मुख्य शीर्षक e.g. चांगला पायाभूत विकास',
  `slug` VARCHAR(150) NOT NULL UNIQUE COMMENT 'URL slug e.g. changla-payabhut-vikas',
  `icon_image` VARCHAR(500) NULL COMMENT 'अपलोड केलेल्या लोगो/आयकॉनचा URL',
  `short_description` VARCHAR(500) NULL COMMENT 'थोडक्यात माहिती / Subtitle',
  `button_text` VARCHAR(100) NOT NULL DEFAULT 'अधिक माहिती पहा',
  `content` LONGTEXT NOT NULL COMMENT 'Word-style rich HTML document formatted from Admin Editor',
  `content_format` ENUM('html', 'json') NOT NULL DEFAULT 'html',
  `display_order` INT NOT NULL DEFAULT 0 COMMENT 'क्रमानुसार मांडणीसाठी क्रम',
  `status` ENUM('active', 'inactive', 'draft') NOT NULL DEFAULT 'active',
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX `idx_sankalp_status_order` (`status`, `display_order`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```

3.4 Admin Panel CMS Endpoints (For Election Admin CRM)
----------------------------------------------------------------------------------------

### 1. `POST /api/v1/admin/sankalp` (Create New Sankalp)
- Headers:
  `Authorization: Bearer <ADMIN_JWT_TOKEN>`
  `Content-Type: multipart/form-data` OR `application/json`

- Body Parameters:
  * `title` (string, required): मराठी नाव e.g., "स्वच्छ व सुंदर कोपरगाव"
  * `slug` (string, optional): e.g., "swachh-va-sundar-kopargaon" (auto-generated if omitted)
  * `short_description` (string, optional): Card subtitle
  * `button_text` (string, optional): Default "अधिक माहिती पहा"
  * `content` (string HTML, required): Sanitized Word-style HTML document
  * `display_order` (integer, optional): e.g., 3
  * `status` (string, optional): 'active' | 'inactive' | 'draft'
  * `icon_image` (file upload): Image file (PNG/JPG/WEBP)

### 2. `PUT /api/v1/admin/sankalp/:id` (Update Existing Sankalp)
Updates all fields or partial fields, including updating the rich HTML content.

### 3. `DELETE /api/v1/admin/sankalp/:id` (Delete or Soft-Delete)
Removes the pledge from public listing.

4. COMPLETE NODE.JS / EXPRESS.JS CONTROLLER & ROUTER IMPLEMENTATION
========================================================================================

Place this code inside your backend directory:

### File: `controllers/sankalpController.js`
```javascript
const db = require('../config/db'); // Your MySQL pool connection

// Helper: Slugify string
function slugify(text) {
  return text
    .toString()
    .toLowerCase()
    .trim()
    .replace(/\s+/g, '-')
    .replace(/[^\w\u0900-\u097F\-]+/g, '')
    .replace(/\-\-+/g, '-');
}

exports.getActiveCards = async (req, res) => {
  try {
    const query = `
      SELECT id, title, slug, icon_image, short_description, button_text, display_order
      FROM pramukh_sankalp
      WHERE status = 'active'
      ORDER BY display_order ASC, id ASC
    `;
    const [rows] = await db.query(query);

    return res.status(200).json({
      success: true,
      message: 'Active Sankalp cards fetched successfully',
      data: rows
    });
  } catch (error) {
    console.error('Error fetching active Sankalp cards:', error);
    return res.status(500).json({
      success: false,
      message: 'Internal server error while fetching Sankalp cards',
      data: []
    });
  }
};

exports.getDetailBySlugOrId = async (req, res) => {
  try {
    const { slugOrId } = req.params;
    if (!slugOrId) {
      return res.status(400).json({ success: false, message: 'Slug or ID is required' });
    }

    const query = `
      SELECT id, title, slug, icon_image, short_description, button_text, 
             content, content_format, display_order, status, created_at, updated_at
      FROM pramukh_sankalp
      WHERE (slug = ? OR id = ?) AND status = 'active'
      LIMIT 1
    `;
    const [rows] = await db.query(query, [slugOrId, slugOrId]);

    if (!rows || rows.length === 0) {
      return res.status(404).json({
        success: false,
        message: 'Sankalp not found or inactive',
        data: null
      });
    }

    return res.status(200).json({
      success: true,
      message: 'Sankalp details fetched successfully',
      data: rows[0]
    });
  } catch (error) {
    console.error('Error fetching Sankalp detail:', error);
    return res.status(500).json({
      success: false,
      message: 'Internal server error while fetching Sankalp details',
      data: null
    });
  }
};

// Admin: Create Sankalp
exports.createSankalp = async (req, res) => {
  try {
    const { title, slug, short_description, button_text, content, display_order, status } = req.body;
    if (!title || !content) {
      return res.status(422).json({ success: false, message: 'Title and content are required' });
    }

    const finalSlug = slug ? slugify(slug) : slugify(title);
    let icon_image = req.body.icon_image || null;

    if (req.file) {
      icon_image = `${req.protocol}://${req.get('host')}/public/uploads/${req.file.filename}`;
    }

    const insertQuery = `
      INSERT INTO pramukh_sankalp 
      (title, slug, icon_image, short_description, button_text, content, display_order, status)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    `;
    const [result] = await db.query(insertQuery, [
      title.trim(),
      finalSlug,
      icon_image,
      short_description ? short_description.trim() : null,
      button_text ? button_text.trim() : 'अधिक माहिती पहा',
      content,
      display_order ? parseInt(display_order, 10) : 0,
      status || 'active'
    ]);

    return res.status(201).json({
      success: true,
      message: 'Sankalp created successfully',
      data: { id: result.insertId, slug: finalSlug }
    });
  } catch (error) {
    console.error('Error creating Sankalp:', error);
    return res.status(500).json({ success: false, message: 'Failed to create Sankalp' });
  }
};

// Admin: Update Sankalp
exports.updateSankalp = async (req, res) => {
```

### File: `middleware/uploadMiddleware.js`
```javascript
const multer = require('multer');
const path = require('path');
const fs = require('fs');

const uploadDir = path.join(__dirname, '../public/uploads');
if (!fs.existsSync(uploadDir)) {
  fs.mkdirSync(uploadDir, { recursive: true });
}

const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    cb(null, uploadDir);
  },
  filename: (req, file, cb) => {
    const ext = path.extname(file.originalname);
    const cleanBaseName = path.basename(file.originalname, ext).replace(/[^\w]/g, '_');
    cb(null, `${cleanBaseName}-${Date.now()}${ext}`);
  }
});

const fileFilter = (req, file, cb) => {
  const allowed = /jpeg|jpg|png|webp|svg/;
  const isExt = allowed.test(path.extname(file.originalname).toLowerCase());
  const isMime = allowed.test(file.mimetype);
  if (isExt && isMime) {
    return cb(null, true);
  }
  cb(new Error('Only JPG, PNG, WEBP, and SVG image files are allowed!'));
};

const upload = multer({
  storage,
  limits: { fileSize: 5 * 1024 * 1024 }, // 5 MB max
  fileFilter
});

module.exports = upload;
```
