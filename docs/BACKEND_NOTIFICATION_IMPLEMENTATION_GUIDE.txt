Sachin Malik Candidate Application (Kopargaon Constituency)
Complete Guide to Fix 404 Notification Click Errors & Standardize Routing
================================================================================

1. PROBLEM SUMMARY & ROOT CAUSE
--------------------------------------------------------------------------------
In the mobile application (React Native / Expo Router), screens are defined
using file-based routing with dedicated naming conventions (e.g. news-detail.tsx,
issue-detail.tsx, notification-detail.tsx).

When notifications were dispatched with REST-style routes such as:
  targetRoute: "/news/4"
Expo Router attempted to look for a screen file at "/app/news/[id].tsx" or
"/app/news/4.tsx". Since neither existed, Expo Router triggered:
  "Unmatched Route: This screen doesn't exist. (404) - /news/4"

To fix this completely, the backend and admin panel must follow the standard
route mapping specification detailed below, while the mobile app has also been
fortified with a universal route resolver and a +not-found fallback interceptor.

/**
 * Normalizes notification targets into official Expo Router paths.
 * Guarantees that the app will never receive a 404 route.
 */
function buildTargetRoute(category, itemId) {
  const cleanId = itemId ? String(itemId).trim() : null;

  switch (category) {
    case 'news':
      return cleanId ? `/news-detail` : `/news`;

    case 'issues':
    case 'issue':
      return cleanId ? `/issue-detail` : `/issues`;

    case 'events':
    case 'event':
      return cleanId ? `/event-detail` : `/events`;

    case 'resolutions':
    case 'resolution':
    case 'sankalp':
      return cleanId ? `/resolution-detail` : `/resolutions`;

    case 'civic-services':
      return `/civic-services`;

    case 'party-membership':
      return `/party-membership`;

    case 'general':
    case 'notifications':
    default:
      return cleanId ? `/notification-detail` : `/notifications`;
  }
}

/**
 * Validates and sanitizes a custom target route entered by an admin.
 */
function sanitizeTargetRoute(rawRoute, fallbackCategory, itemId) {
  if (!rawRoute || typeof rawRoute !== 'string' || rawRoute.trim() === '') {
    return buildTargetRoute(fallbackCategory, itemId);
  }

  let clean = rawRoute.trim();

  // If admin typed "/news/4", normalize to "/news-detail"
  const newsMatch = clean.match(/^\/?news\/([^\/]+)$/i);
  if (newsMatch) {
    return { targetRoute: '/news-detail', id: newsMatch[1] };
  }

  // If admin typed "/issues/12", normalize to "/issue-detail"
  const issueMatch = clean.match(/^\/?(?:issues|issue)\/([^\/]+)$/i);
  if (issueMatch) {
    return { targetRoute: '/issue-detail', id: issueMatch[1] };
  }

  // If admin typed "/events/5", normalize to "/event-detail"
  const eventMatch = clean.match(/^\/?(?:events|event)\/([^\/]+)$/i);
  if (eventMatch) {
    return { targetRoute: '/event-detail', id: eventMatch[1] };
  }

  // If admin typed "/resolutions/1", normalize to "/resolution-detail"
  const resMatch = clean.match(/^\/?(?:resolutions|resolution|sankalp)\/([^\/]+)$/i);
  if (resMatch) {
    return { targetRoute: '/resolution-detail', id: resMatch[1] };
  }

  // If admin typed "notifications" or "/notifications"
  if (clean === '/notifications' || clean === 'notifications') {
    return { targetRoute: '/notifications', id: null };
  }

  // Ensure leading slash
  if (!clean.startsWith('/')) {
    clean = '/' + clean;
  }

  return { targetRoute: clean, id: itemId ? String(itemId) : null };
}

module.exports = {
  buildTargetRoute,
  sanitizeTargetRoute,
};

const admin = require('firebase-admin'); // Firebase Admin SDK
const { Expo } = require('expo-server-sdk'); // Expo Server SDK
const db = require('../config/database'); // Your database connection pool
const { sanitizeTargetRoute } = require('./notificationHelper');

const expo = new Expo();

/**
 * Dispatches a push notification to devices and saves it to the database.
 * 
 * @param {Object} params
 * @param {string} params.title - Notification title
 * @param {string} params.description - Notification summary
 * @param {string} [params.category] - 'news' | 'issues' | 'events' | 'resolutions' | 'general'
 * @param {string} [params.targetRoute] - Target screen route
 * @param {string|number} [params.itemId] - ID of the target resource (e.g. newsId, trackingId)
 * @param {string} [params.mediaType] - 'none' | 'image' | 'video' | 'pdf'
 * @param {string} [params.mediaUrl] - Image or document URL
 * @param {string} [params.userId] - Optional: specific user ID for targeted dispatch
 * @param {string} [params.mobile] - Optional: specific citizen phone number
 */
async function sendNotification(params) {
  const {
    title,
    description,
    category = 'general',
    targetRoute: rawTargetRoute,
    itemId,
    mediaType = 'none',
    mediaUrl = null,
    userId = null,
    mobile = null,
    priority = 'normal',
  } = params;

  // 1. Sanitize the target route so 404 is impossible
  const { targetRoute, id: resolvedId } = sanitizeTargetRoute(rawTargetRoute, category, itemId);

  // 2. Insert record into `notifications` database table
  const insertQuery = `
    INSERT INTO notifications 
      (category, title, description, message, target_route, item_id, media_type, media_url, priority, sent_at)
    VALUES 
      (?, ?, ?, ?, ?, ?, ?, ?, ?, NOW())
  `;

  let notificationDbId = null;
  try {
    const [result] = await db.execute(insertQuery, [
      category,
      title,
      description,
      description,
      targetRoute,
      resolvedId || null,
      mediaType,
      mediaUrl,
      priority,
    ]);
    notificationDbId = result.insertId;
  } catch (dbErr) {
    console.error('Error inserting notification into database:', dbErr);
  }

  // 3. Prepare FCM / Expo payload data
  // NOTE: In FCM data payload, all values must be strings!
  const pushDataPayload = {
    targetRoute: String(targetRoute),
    id: String(resolvedId || notificationDbId || 'push-item'),
    title: String(title),
    message: String(description),
    body: String(description),
    category: String(category),
    mediaType: String(mediaType || 'none'),
    mediaUrl: mediaUrl ? String(mediaUrl) : '',
    notificationId: notificationDbId ? String(notificationDbId) : '',
  };

  // If targeting specific issue or news item, include explicit keys for safety
  if (category === 'news' && resolvedId) {
    pushDataPayload.newsId = String(resolvedId);
  } else if ((category === 'issues' || category === 'issue') && resolvedId) {
    pushDataPayload.issueId = String(resolvedId);
    pushDataPayload.trackingId = String(resolvedId);
  } else if (category === 'events' && resolvedId) {
    pushDataPayload.eventId = String(resolvedId);
  } else if (category === 'resolutions' && resolvedId) {
    pushDataPayload.resolutionId = String(resolvedId);
  }

  // 4. Retrieve target device tokens
  let tokensQuery = 'SELECT token, fcm_token, expo_token FROM device_tokens WHERE is_active = 1';
  const queryParams = [];

  if (userId) {
    tokensQuery += ' AND user_id = ?';
    queryParams.push(userId);
  } else if (mobile) {
    const cleanMobile = String(mobile).replace(/\D/g, '').slice(-10);
    tokensQuery += ' AND mobile = ?';
    queryParams.push(cleanMobile);
  }

  let deviceRows = [];
  try {
    const [rows] = await db.execute(tokensQuery, queryParams);
    deviceRows = rows;
  } catch (err) {
    console.error('Error fetching device tokens:', err);
  }

  if (deviceRows.length === 0) {
    console.log('No registered devices found for dispatch.');
    return { success: true, count: 0, notificationId: notificationDbId };
  }

  // 5. Group into FCM and Expo Push tokens
  const fcmTokens = [];
  const expoTokens = [];

  deviceRows.forEach((row) => {
    const fcm = row.fcm_token;
    const expoTok = row.expo_token || row.token;

    if (fcm && typeof fcm === 'string' && fcm.length > 20) {
      fcmTokens.push(fcm);
    } else if (expoTok && Expo.isExpoPushToken(expoTok)) {
      expoTokens.push(expoTok);
    }
  });

  // 6. Dispatch via Firebase Cloud Messaging (FCM)
  if (fcmTokens.length > 0) {
    try {
      // Chunk tokens in batches of 500 (FCM limit)
      const chunkSize = 500;
      for (let i = 0; i < fcmTokens.length; i += chunkSize) {
        const batchTokens = fcmTokens.slice(i, i + chunkSize);
        const fcmMessage = {
          notification: {
            title: title,
            body: description,
          },
          data: pushDataPayload,
          android: {
            priority: priority === 'high' ? 'high' : 'normal',
            notification: {
              channelId: 'default',
              color: '#EA580C',
              sound: 'default',
              clickAction: 'FLUTTER_NOTIFICATION_CLICK', // standard Android intent trigger
            },
          },
          tokens: batchTokens,
        };

        const response = await admin.messaging().sendEachForMulticast(fcmMessage);
        console.log(`✅ FCM Sent: ${response.successCount} success, ${response.failureCount} failed`);

        // Handle invalid/expired tokens cleanup
        if (response.failureCount > 0) {
          response.responses.forEach((resp, idx) => {
            if (!resp.success) {
              const errCode = resp.error?.code;
              if (
                errCode === 'messaging/invalid-registration-token' ||
                errCode === 'messaging/registration-token-not-registered'
              ) {
                const badToken = batchTokens[idx];
                db.execute('DELETE FROM device_tokens WHERE fcm_token = ? OR token = ?', [badToken, badToken]).catch(() => {});
              }
            }
          });
        }
      }
    } catch (fcmErr) {
      console.error('Error during FCM multicast dispatch:', fcmErr);
    }
  }

  // 7. Dispatch via Expo Push Service
  if (expoTokens.length > 0) {
    try {
      const messages = expoTokens.map((pushToken) => ({
        to: pushToken,
        sound: 'default',
        title: title,
        body: description,
        data: pushDataPayload,
        channelId: 'default',
      }));

      const chunks = expo.chunkPushNotifications(messages);
      for (const chunk of chunks) {
        try {
          const ticketChunk = await expo.sendPushNotificationsAsync(chunk);
          console.log(`✅ Expo Push Tickets generated: ${ticketChunk.length}`);
        } catch (error) {
          console.error('Error sending Expo push chunk:', error);
        }
      }
    } catch (expoErr) {
      console.error('Error during Expo Push dispatch:', expoErr);
    }
  }

  return {
    success: true,
    notificationId: notificationDbId,
    targetRoute,
    fcmCount: fcmTokens.length,
    expoCount: expoTokens.length,
  };
}

module.exports = {
  sendNotification,
};

const { sendNotification } = require('../services/notificationService');
const db = require('../config/database');

/**
 * POST /api/v1/notifications/send
 * Called from Admin Panel to dispatch notifications.
 */
async function handleSendNotification(req, res) {
  try {
    const {
      title,
      description,
      category,
      targetRoute,
      itemId,
      mediaType,
      mediaUrl,
      userId,
      mobile,
      priority,
    } = req.body;

    if (!title || !description) {
      return res.status(400).json({
        success: false,
        message: 'Title and description are required.',
      });
    }

    const result = await sendNotification({
      title,
      description,
      category,
      targetRoute,
      itemId,
      mediaType,
      mediaUrl,
      userId,
      mobile,
      priority,
    });

    return res.status(200).json({
      success: true,
      message: 'Notification dispatched successfully',
      data: result,
    });
  } catch (error) {
    console.error('handleSendNotification error:', error);
    return res.status(500).json({
      success: false,
      message: 'Failed to send notification',
      error: error.message,
    });
  }
}

/**
 * POST /api/v1/notifications/register-token
 * Called by mobile app on startup or login to register push token.
 */
async function handleRegisterToken(req, res) {
  try {
    const {
      token,
      fcmToken,
      fcm_token,
      expoToken,
      expo_token,
      platform = 'android',
      userId,
      user_id,
      mobile,
      phone,
      name,
      candidateId,
      candidate_id,
    } = req.body;

    const finalToken = token || fcmToken || fcm_token || expoToken || expo_token;
    if (!finalToken) {
      return res.status(400).json({ success: false, message: 'Push token required.' });
    }

    const cleanMobile = (mobile || phone) ? String(mobile || phone).replace(/\D/g, '').slice(-10) : null;
    const cleanUserId = userId || user_id || null;
    const finalFcm = fcmToken || fcm_token || (finalToken.length > 50 ? finalToken : null);
    const finalExpo = expoToken || expo_token || (finalToken.startsWith('ExponentPushToken') ? finalToken : null);

    const upsertQuery = `
      INSERT INTO device_tokens 
        (token, fcm_token, expo_token, platform, user_id, mobile, name, candidate_id, is_active, last_active_at)
      VALUES 
        (?, ?, ?, ?, ?, ?, ?, ?, 1, NOW())
      ON DUPLICATE KEY UPDATE
        fcm_token = COALESCE(VALUES(fcm_token), fcm_token),
        expo_token = COALESCE(VALUES(expo_token), expo_token),
        user_id = COALESCE(VALUES(user_id), user_id),
        mobile = COALESCE(VALUES(mobile), mobile),
        name = COALESCE(VALUES(name), name),
        is_active = 1,
        last_active_at = NOW()
    `;

    await db.execute(upsertQuery, [
      finalToken,
      finalFcm,
      finalExpo,
      platform,
      cleanUserId,
      cleanMobile,
      name || null,
      candidateId || candidate_id || 'sachin-malik',
    ]);

    return res.status(200).json({
      success: true,
      message: 'Push token registered successfully',
    });
  } catch (error) {
    console.error('handleRegisterToken error:', error);
    return res.status(500).json({
      success: false,
      message: 'Failed to register token',
    });
  }
}

/**
 * GET /api/v1/notifications
 * Lists all active notifications for mobile app sync.
 */
async function handleGetNotifications(req, res) {
  try {
    const page = Math.max(1, parseInt(req.query.page || '1', 10));
    const limit = Math.min(100, Math.max(1, parseInt(req.query.limit || '30', 10)));
    const offset = (page - 1) * limit;

    const [rows] = await db.execute(
      `SELECT 
        id, category, icon, icon_color AS iconColor, icon_bg AS iconBg,
        title, description, message, indicator_dot AS indicatorDot,
        target_route AS targetRoute, item_id AS itemId,
        media_type AS mediaType, media_url AS mediaUrl,
        priority, sent_at AS sentAt, is_active AS isActive,
        created_at AS createdAt
      FROM notifications
      WHERE is_active = 1
      ORDER BY sent_at DESC, id DESC
      LIMIT ? OFFSET ?`,
      [String(limit), String(offset)]
    );

    // Format human-friendly time for Marathi UI
    const mapped = rows.map((r) => {
      // Ensure targetRoute has no broken REST style
      let sanitized = r.targetRoute;
      if (sanitized && sanitized.startsWith('/news/') && r.itemId) {
        sanitized = '/news-detail';
      }
      return {
        ...r,
        targetRoute: sanitized,
        timeAgo: 'आत्ताच',
      };
    });

    const [countRows] = await db.execute(
      'SELECT COUNT(*) as total FROM notifications WHERE is_active = 1'
    );
    const total = countRows[0]?.total || 0;

    return res.status(200).json({
      success: true,
      data: mapped,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    });
  } catch (error) {
    console.error('handleGetNotifications error:', error);
    return res.status(500).json({ success: false, message: 'Server error' });
  }
}
