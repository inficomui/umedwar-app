1. ARCHITECTURAL OVERVIEW & MOBILE APP INTEGRATION
================================================================================

The Mobile App displays an interactive horizontal swipe strip of Political Campaign
Banners right on the Home Page ("Section: Political Campaign Banners").

Each card in this section features:
  * `id`: Unique identifier (e.g. 1, 2, "pb-1")
  * `bg`: Distinctive vibrant background color (e.g. #C2410C, #1D4ED8, #065F46)
  * `title`: Primary Marathi campaign heading (e.g. "आपला हक्काचा माणूस,\nसचिन मलिक!")
  * `subtitle`: Detailed message or pledge (e.g. "जनसेवेसाठी सदैव तत्पर, कोपरगावच्या सर्वांगीण विकासासाठी कटिबद्ध!")
  * `badge`: Top-left pill label (e.g. "जनसेवा मोहीम २०२६", "विकास अजेंडा")
  * `rightLabel`: Top-right label/party (e.g. "लोकनेता", "विकास", "शेती")
  * `icon`: Center icon name (Material Community / Paper icon e.g. "account-group", "bullhorn", "city-variant-outline", "sprout")
  * `ctaText`: Action button label (e.g. "अधिक जाणून घ्या")
  * `ctaRoute`: Navigation target inside the mobile app (e.g. "/about", "/resolutions", "/news", "/contact")
  * `sortOrder`: Display order integer
  * `isActive`: Boolean toggle (1 = visible in app, 0 = hidden)

MOBILE APP CONSUMPTION:
The mobile app automatically pulls active banners from the existing unified endpoint:
  GET https://backend.misachinmalik.in/api/v1/app-config

Under the response key: `data.politicalBanners` (or `data.political_banners`).
If the backend does not return banners, the mobile app falls back to the default seed banners.

2.1 RAW SQL TABLE CREATION SCRIPT:
--------------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS `political_banners` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `candidate_id` VARCHAR(100) NOT NULL DEFAULT 'sachin-malik',
  `bg_color` VARCHAR(25) NOT NULL DEFAULT '#C2410C',
  `title` VARCHAR(255) NOT NULL,
  `subtitle` TEXT NOT NULL,
  `badge` VARCHAR(100) NOT NULL DEFAULT 'जनसेवा मोहीम २०२६',
  `icon` VARCHAR(100) NOT NULL DEFAULT 'account-group',
  `right_label` VARCHAR(100) NOT NULL DEFAULT 'लोकनेता',
  `cta_text` VARCHAR(100) NOT NULL DEFAULT 'अधिक जाणून घ्या',
  `cta_route` VARCHAR(100) NOT NULL DEFAULT '/about',
  `sort_order` INT NOT NULL DEFAULT 1,
  `is_active` TINYINT(1) NOT NULL DEFAULT 1,
  `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  INDEX `idx_pb_candidate` (`candidate_id`),
  INDEX `idx_pb_active_order` (`is_active`, `sort_order`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

4.1 CONTROLLER (`src/controllers/politicalBanner.controller.ts`):
--------------------------------------------------------------------------------
import { Request, Response } from 'express';
import { db } from '../config/database';

export class PoliticalBannerController {
  /**
   * Helper to fetch active banners for public app-config
   */
  static async getActiveBannersForApp(candidateId: string = 'sachin-malik') {
    const [rows]: any = await db.query(
      `SELECT id, bg_color as bg, title, subtitle, badge, icon, 
              right_label as rightLabel, cta_text as ctaText, 
              cta_route as ctaRoute, sort_order as sortOrder, is_active as isActive
       FROM political_banners
       WHERE candidate_id = ? AND is_active = 1
       ORDER BY sort_order ASC, id ASC`,
      [candidateId]
    );
    return rows;
  }

  /**
   * Admin: List all banners
   * GET /api/v1/admin/political-banners
   */
  static async getAdminBanners(req: Request, res: Response) {
    try {
      const status = req.query.status as string;
      const search = (req.query.search as string || '').trim();

      let whereClauses: string[] = ['candidate_id = ?'];
      let queryParams: any[] = ['sachin-malik'];

      if (status === 'active') {
        whereClauses.push('is_active = 1');
      } else if (status === 'inactive') {
        whereClauses.push('is_active = 0');
      }

      if (search) {
        whereClauses.push('(title LIKE ? OR subtitle LIKE ? OR badge LIKE ?)');
        const searchLike = `%${search}%`;
        queryParams.push(searchLike, searchLike, searchLike);
      }

      const whereSql = `WHERE ${whereClauses.join(' AND ')}`;

      const [rows]: any = await db.query(
        `SELECT id, candidate_id, bg_color, title, subtitle, badge, icon,
                right_label, cta_text, cta_route, sort_order, is_active,
                created_at, updated_at
         FROM political_banners
         ${whereSql}
         ORDER BY sort_order ASC, id ASC`,
        queryParams
      );

      return res.json({
        success: true,
        data: {
          banners: rows,
          total: rows.length,
        },
      });
    } catch (error: any) {
      console.error('[PoliticalBannerController] getAdminBanners error:', error);
      return res.status(500).json({ success: false, message: 'Server error' });
    }
  }

  /**
   * Admin: Create new banner
   * POST /api/v1/admin/political-banners
   */
  static async createBanner(req: Request, res: Response) {
    try {
      const b = req.body;
      const candidateId = b.candidateId || b.candidate_id || 'sachin-malik';
      const bg = b.bg || b.bg_color || '#C2410C';
      const title = (b.title || '').trim();
      const subtitle = (b.subtitle || '').trim();
      const badge = (b.badge || 'जनसेवा मोहीम २०२६').trim();
      const icon = (b.icon || 'account-group').trim();
      const rightLabel = (b.rightLabel || b.right_label || 'लोकनेता').trim();
      const ctaText = (b.ctaText || b.cta_text || 'अधिक जाणून घ्या').trim();
      const ctaRoute = (b.ctaRoute || b.cta_route || '/about').trim();
      const sortOrder = parseInt(b.sortOrder ?? b.sort_order ?? 1) || 1;
      const isActive = b.isActive ?? b.is_active ?? true ? 1 : 0;

      if (!title) {
        return res.status(400).json({ success: false, message: 'शीर्षक (Title) भरणे आवश्यक आहे.' });
      }
      if (!subtitle) {
        return res.status(400).json({ success: false, message: 'उपशीर्षक (Subtitle) भरणे आवश्यक आहे.' });
      }

      const [result]: any = await db.query(
        `INSERT INTO political_banners 
         (candidate_id, bg_color, title, subtitle, badge, icon, right_label, cta_text, cta_route, sort_order, is_active)
         VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
        [candidateId, bg, title, subtitle, badge, icon, rightLabel, ctaText, ctaRoute, sortOrder, isActive]
      );

      return res.status(201).json({
        success: true,
        message: 'प्रचार बॅनर यशस्वीरीत्या तयार केला गेला आहे.',
        data: {
          id: result.insertId,
          bg,
          title,
          subtitle,
          badge,
          icon,
          rightLabel,
          ctaText,
          ctaRoute,
          sortOrder,
          isActive: Boolean(isActive),
        },
      });
    } catch (error: any) {
      console.error('[PoliticalBannerController] createBanner error:', error);
      return res.status(500).json({ success: false, message: 'Server error' });
    }
  }

  /**
   * Admin: Update banner
   * PUT /api/v1/admin/political-banners/:id
   */
  static async updateBanner(req: Request, res: Response) {
    try {
      const { id } = req.params;
      const b = req.body;
      const bg = b.bg || b.bg_color || '#C2410C';
      const title = (b.title || '').trim();
      const subtitle = (b.subtitle || '').trim();
      const badge = (b.badge || 'जनसेवा मोहीम २०२६').trim();
      const icon = (b.icon || 'account-group').trim();
      const rightLabel = (b.rightLabel || b.right_label || 'लोकनेता').trim();
      const ctaText = (b.ctaText || b.cta_text || 'अधिक जाणून घ्या').trim();
      const ctaRoute = (b.ctaRoute || b.cta_route || '/about').trim();
      const sortOrder = parseInt(b.sortOrder ?? b.sort_order ?? 1) || 1;
      const isActive = b.isActive ?? b.is_active ?? true ? 1 : 0;

      if (!title || !subtitle) {
        return res.status(400).json({ success: false, message: 'Title आणि Subtitle दोन्ही आवश्यक आहेत.' });
      }

      const [result]: any = await db.query(
        `UPDATE political_banners 
         SET bg_color = ?, title = ?, subtitle = ?, badge = ?, icon = ?, 
             right_label = ?, cta_text = ?, cta_route = ?, sort_order = ?, is_active = ?, updated_at = NOW()
         WHERE id = ?`,
        [bg, title, subtitle, badge, icon, rightLabel, ctaText, ctaRoute, sortOrder, isActive, id]
      );

      if (result.affectedRows === 0) {
        return res.status(404).json({ success: false, message: 'Banner not found.' });
      }

      return res.json({
        success: true,
        message: 'प्रचार बॅनर यशस्वीरीत्या अद्ययावत (Update) झाला आहे.',
      });
    } catch (error: any) {
      console.error('[PoliticalBannerController] updateBanner error:', error);
      return res.status(500).json({ success: false, message: 'Server error' });
    }
  }

  /**
   * Admin: Toggle status
   * PATCH /api/v1/admin/political-banners/:id/status
   */
  static async toggleStatus(req: Request, res: Response) {
    try {
      const { id } = req.params;
      const { isActive } = req.body;
      const activeVal = isActive ? 1 : 0;

      await db.query(
        `UPDATE political_banners SET is_active = ?, updated_at = NOW() WHERE id = ?`,
        [activeVal, id]
      );

      return res.json({ success: true, message: 'स्थिती यशस्वीरीत्या बदलली.' });
    } catch (error: any) {
      return res.status(500).json({ success: false, message: 'Server error' });
    }
  }

  /**
   * Admin: Delete banner
   * DELETE /api/v1/admin/political-banners/:id
   */
  static async deleteBanner(req: Request, res: Response) {
    try {
      const { id } = req.params;
      await db.query(`DELETE FROM political_banners WHERE id = ?`, [id]);
      return res.json({ success: true, message: 'बॅनर यशस्वीरीत्या हटवला गेला.' });
    } catch (error: any) {
      return res.status(500).json({ success: false, message: 'Server error' });
    }
  }

  /**
   * Admin: Batch reorder banners
   * POST /api/v1/admin/political-banners/reorder
   */
  static async reorderBanners(req: Request, res: Response) {
    try {
      const { order } = req.body; // Array of { id, sortOrder }
      if (Array.isArray(order)) {
        for (const item of order) {
          await db.query(`UPDATE political_banners SET sort_order = ? WHERE id = ?`, [
            item.sortOrder,
            item.id,
          ]);
        }
      }
      return res.json({ success: true, message: 'क्रमवारी यशस्वीरीत्या जतन केली.' });
    } catch (error: any) {
      return res.status(500).json({ success: false, message: 'Server error' });
    }
  }
}

5. LARAVEL / PHP CONTROLLER CODE (Alternative Backend Option)

If backend runs on Laravel:

// app/Http/Controllers/Api/PoliticalBannerController.php
namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;
use App\Models\PoliticalBanner;

class PoliticalBannerController extends Controller
{
    // GET /api/v1/admin/political-banners
    public function index(Request $request)
    {
        $query = PoliticalBanner::where('candidate_id', 'sachin-malik');

        if ($request->has('status') && $request->status !== 'all') {
            $query->where('is_active', $request->status === 'active' ? 1 : 0);
        }

        if ($request->filled('search')) {
            $s = '%' . $request->search . '%';
            $query->where(function ($q) use ($s) {
                $q->where('title', 'like', $s)
                  ->orWhere('subtitle', 'like', $s);
            });
        }

        $banners = $query->orderBy('sort_order', 'asc')->get();

        return response()->json([
            'success' => true,
            'data' => [
                'banners' => $banners,
                'total' => $banners->count(),
            ]
        ]);
    }

    // POST /api/v1/admin/political-banners
    public function store(Request $request)
    {
        $validated = $request->validate([
            'title'       => 'required|string|max:255',
            'subtitle'    => 'required|string',
            'bg_color'    => 'nullable|string|max:25',
            'badge'       => 'nullable|string|max:100',
            'icon'        => 'nullable|string|max:100',
            'right_label' => 'nullable|string|max:100',
            'cta_text'    => 'nullable|string|max:100',
            'cta_route'   => 'nullable|string|max:100',
            'sort_order'  => 'nullable|integer',
            'is_active'   => 'nullable|boolean',
        ]);

        $banner = PoliticalBanner::create([
            'candidate_id' => 'sachin-malik',
            'bg_color'     => $validated['bg_color'] ?? '#C2410C',
            'title'        => $validated['title'],
            'subtitle'     => $validated['subtitle'],
            'badge'        => $validated['badge'] ?? 'जनसेवा मोहीम २०२६',
            'icon'         => $validated['icon'] ?? 'account-group',
            'right_label'  => $validated['right_label'] ?? 'लोकनेता',
            'cta_text'     => $validated['cta_text'] ?? 'अधिक जाणून घ्या',
            'cta_route'    => $validated['cta_route'] ?? '/about',
            'sort_order'   => $validated['sort_order'] ?? 1,
            'is_active'    => $validated['is_active'] ?? true,
        ]);

        return response()->json([
            'success' => true,
            'message' => 'प्रचार बॅनर तयार झाला.',
            'data'    => $banner,
        ], 201);
    }

    // PUT /api/v1/admin/political-banners/{id}
    public function update(Request $request, $id)
    {
        $banner = PoliticalBanner::findOrFail($id);
        $banner->update($request->all());

        return response()->json([
            'success' => true,
            'message' => 'प्रचार बॅनर अद्ययावत झाला.',
            'data'    => $banner,
        ]);
    }

    // DELETE /api/v1/admin/political-banners/{id}
    public function destroy($id)
    {
        PoliticalBanner::destroy($id);
        return response()->json(['success' => true, 'message' => 'बॅनर हटवला गेला.']);
    }
}
